# M2.2 - postavitev prizorisca za reprodukcijo R1. Ukaze poslje .\r1-run.ps1 na
# standardni vhod serverja; rocno lepljenje v konzolo je nezanesljivo (M0.6, past 1).
#
# Dve vzporedni progi, enaki v vsem razen v jahacih:
#   proga M (x = -8..8)   8 nosilcev Z jahaci   - merjeno stanje
#   proga S (x = 16..32)  8 nosilcev BREZ njih  - kontrola
# Brez kontrole razlika ne pomeni nicesar; razlogi v docs/scenariji/M2.2-R1.md.
#
# Postavitev vzdolz +Z, superflat, povrsina y=3, entitete stojijo na y=4:
#   z=12 start   z=20 stopnica   z=28 zid z enimi vrati   z=36 stopnica   z=44 cilj
# Razdalja start -> cilj je 32 blokov.
#
# Ukazi so idempotentni: 'noppes slay npcs' postavi isDead=true (CmdSlay.java:140),
# medtem ko bi '/kill' CustomNPC-ja samo ubil in bi po restartu ozivel (M0.6, past 3).

# --- determinizem ---
gamerule doDaylightCycle false
gamerule doWeatherCycle false
gamerule doMobSpawning false
gamerule doFireTick false
gamerule mobGriefing false
gamerule randomTickSpeed 0
gamerule announceAdvancements false
gamerule spectatorsGenerateChunks false
time set 6000
weather clear 1000000

# --- ocisti prejsnji zagon ---
noppes slay npcs

# --- proga M: stopnici ---
fill -8 4 20 8 4 20 minecraft:stone
fill -8 4 36 8 4 36 minecraft:stone

# --- proga M: zid z enimi vrati pri x=0 ---
# Vrata so edini prehod. Oak door se v 1.12.2 registrira kot 'wooden_door'
# (Blocks.java:392), ne 'oak_door'. Spodnja polovica meta 3 = FACING NORTH
# (BlockDoor.getStateFromMeta, :423: getHorizontal(meta&3).rotateYCCW()),
# zgornja meta 8 = UPPER, tecaj levo, brez signala.
fill -8 4 28 8 6 28 minecraft:stone
fill 0 4 28 0 6 28 minecraft:air
setblock 0 4 28 minecraft:wooden_door 3
setblock 0 5 28 minecraft:wooden_door 8

# --- proga S: enako, zamaknjeno za x+24 ---
fill 16 4 20 32 4 20 minecraft:stone
fill 16 4 36 32 4 36 minecraft:stone
fill 16 4 28 32 6 28 minecraft:stone
fill 24 4 28 24 6 28 minecraft:air
setblock 24 4 28 minecraft:wooden_door 3
setblock 24 5 28 minecraft:wooden_door 8

# --- proga M: 8 nosilcev in 8 jahacev ---
# Jahaci so 2 bloka za nosilci pri istem x; skripta ju pari po x, zato je
# dodeljevanje deterministicno.
noppes clone spawn R1_Carrier 1 -7,4,12
noppes clone spawn R1_Carrier 1 -5,4,12
noppes clone spawn R1_Carrier 1 -3,4,12
noppes clone spawn R1_Carrier 1 -1,4,12
noppes clone spawn R1_Carrier 1 1,4,12
noppes clone spawn R1_Carrier 1 3,4,12
noppes clone spawn R1_Carrier 1 5,4,12
noppes clone spawn R1_Carrier 1 7,4,12
noppes clone spawn R1_Rider 1 -7,4,14
noppes clone spawn R1_Rider 1 -5,4,14
noppes clone spawn R1_Rider 1 -3,4,14
noppes clone spawn R1_Rider 1 -1,4,14
noppes clone spawn R1_Rider 1 1,4,14
noppes clone spawn R1_Rider 1 3,4,14
noppes clone spawn R1_Rider 1 5,4,14
noppes clone spawn R1_Rider 1 7,4,14

# --- proga S: 8 nosilcev brez jahacev ---
noppes clone spawn R1_Carrier 1 17,4,12
noppes clone spawn R1_Carrier 1 19,4,12
noppes clone spawn R1_Carrier 1 21,4,12
noppes clone spawn R1_Carrier 1 23,4,12
noppes clone spawn R1_Carrier 1 25,4,12
noppes clone spawn R1_Carrier 1 27,4,12
noppes clone spawn R1_Carrier 1 29,4,12
noppes clone spawn R1_Carrier 1 31,4,12

# --- cilja napada ---
noppes clone spawn R1_Target 1 0,4,44
noppes clone spawn R1_Target 1 24,4,44

save-all flush

# R1_Control tu NI. Skripta zacne steti takoj ob spawnu, zato ga .\r1-run.ps1 spawna
# sele potem, ko sta pogoja meritve iz M2.1d izpolnjena: chunki prisilno nalozeni in
# ogrevanje koncano. Sicer bi se scenarij zacel pod drugacnim pogojem, kot ga merimo.
